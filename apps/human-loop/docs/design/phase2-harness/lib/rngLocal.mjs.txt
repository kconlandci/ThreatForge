/** Same mulberry32 + murmur3 seeding as lib/game/rng.ts (copied so bots need no engine import). */
export function seedState(seed) {
  let h;
  if (typeof seed === "string") {
    h = 0x811c9dc5;
    for (let i = 0; i < seed.length; i++) {
      h ^= seed.charCodeAt(i);
      h = Math.imul(h, 0x01000193);
    }
  } else h = Number.isFinite(seed) ? Math.trunc(seed) >>> 0 : 0;
  h ^= h >>> 16;
  h = Math.imul(h, 0x85ebca6b);
  h ^= h >>> 13;
  h = Math.imul(h, 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}
export function nextFloat(state) {
  const next = (state + 0x6d2b79f5) >>> 0;
  let t = next;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return { value: ((t ^ (t >>> 14)) >>> 0) / 4294967296, state: next };
}
export function nextInt(state, n) {
  const r = nextFloat(state);
  return { value: Math.floor(r.value * n), state: r.state };
}
